#mockstick_v2
# 高级脚本示例: 演示条件宏 DSL 的全部指令
# 用法: 在"宏与脚本"页的"高级脚本"区载入本文件, 语法检查后运行

# --- 基础操作 ---
tap 540 1170              # 点击屏幕中心
wait 1.5                  # 等待 1.5 秒
swipe 540 1800 540 800 400   # 上滑翻页 (400ms)
key 4                     # 按返回键
random_wait 0.5 2         # 随机等待 0.5~2 秒 (防检测)

# --- 循环: 重复点击 5 次 ---
label click_loop
tap 540 1170
wait 1
goto click_loop 4         # 跳回 4 次 (加上首次共执行 5 轮)

# --- 条件判断: 等待某个像素变成指定颜色 ---
# 等待 (100,200) 处变成红色 FF0000 (容差 30), 最多等 20 秒
wait_pixel 100 200 FF0000 30 20

# 如果 (500,500) 是绿色则跳到 found 标签, 否则继续
if_pixel 500 500 00FF00 30 found

# --- 找图: 在屏幕上找模板并点击 (模板用"自动化"页的截取工具生成) ---
# click_image template01.mstr 0.9 15

# --- OCR: 等待屏幕出现指定文字 (需要 Windows OCR, 见说明页) ---
# wait_text 战斗胜利 30

label found
tap 500 500
stop                      # 结束脚本
